Code example

Solution of a Codewars kata: sum of numbers in an array.

function sum(numbers) {
  if (!Array.isArray(numbers) || numbers.length === 0) {
    return 0;
  }

  let total = 0;

  for (let i = 0; i < numbers.length; i += 1) {
    total += numbers[i];
  }

  return total;
}
